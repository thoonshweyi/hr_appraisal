SELECT 
    e.employee_code AS "Employee Code",
    e.employee_name AS "Name",
    b.branch_name AS "Branch",
    pl.name AS "Position Level"
FROM public.employees e
INNER JOIN public.branches b 
    ON e.branch_id = b.branch_id
INNER JOIN public.position_levels pl
    ON e.position_level_id::int = pl.id
WHERE e.status_id = 1
AND e.position_level_id NOT IN ('1', '12')
ORDER BY e.id DESC;
